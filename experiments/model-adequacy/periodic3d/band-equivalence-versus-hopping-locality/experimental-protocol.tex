\begin{abstract}
This protocol defines a prospective three-dimensional simple-cubic extension of the
publication-independent experiment \emph{Band Equivalence versus Hopping Locality}. It
constructs the finite parent on
$(\mathbb C^N)^{\otimes3}\otimes\mathbb C^4$ with Kronecker translation operators,

spectral subspace, and converts a declared periodic frame to a complete

$\theta_a(\mathbf q)=\tfrac a3\sum_{\alpha=1}^3\sin(2\pi q_\alpha)$ while preserving

fixed sampled band accuracy while separately reporting parent--retained reconstruction,

Monkhorst--Pack actions at $N=8,16,32$ now exercise these channels; they are neither a
qualified convergence result, retained run, material claim, nor continuous-domain

\section{Scientific question}

Band spectra remain invariant under momentum-dependent unitary similarity, while
real-space hopping locality depends on representation. This is established in the
Wannier literature~\cite{marzari1997,marzari2012,marrazzo2024}; compact and maximally
localized representatives need not coincide~\cite{sathe2024}, and hopping truncation can
degrade band and response accuracy~\cite{ghosh2025}.

The experiment asks how a third exchange-equivalent periodic coordinate changes the
relation between complete Bloch-spectral, retained-frame, and Wannier/tight-binding
representations and the cubic-complete hopping support required by two gauge-related
rank-two fields at equal sampled band accuracy. The complete fields remain physically
equivalent under consistent transformation of states and operators. Range inflation
after truncation measures representation cost, not physical inequivalence or
material-model error. Eigenvalue lists are used for spectral comparisons but are never
treated as operators or subtracted from Hamiltonian matrices.

\section{Finite spectral and tight-binding realizations}

For construction size $N$, define
\begin{equation}
 \mathcal L_N^{(3)}=(\mathbb C^N)^{\otimes3},\qquad
 \mathcal H_{p,N}^{(3)}=\mathcal L_N^{(3)}\otimes\mathbb C^4,\qquad
 \mathcal H_{r,N}^{(3)}=\mathcal L_N^{(3)}\otimes\mathbb C^2.
 \label{eq:spaces}
\end{equation}
Let $\mathsf F_N$ be the unitary centered discrete Fourier matrix and define
$\mathsf S_N$ by
\begin{equation}
 \mathsf F_N\mathsf S_N\mathsf F_N^\dagger
 =\operatorname{diag}\left(e^{2\pi iq_0},\ldots,e^{2\pi iq_{N-1}}\right).
\end{equation}
Set
\begin{align}
 \mathsf S_x&=\mathsf S_N\otimes I_N\otimes I_N,&
 \mathsf S_y&=I_N\otimes\mathsf S_N\otimes I_N,\\
 \mathsf S_z&=I_N\otimes I_N\otimes\mathsf S_N,&
 \mathcal F_{3,N}&=\mathsf F_N^{\otimes3}.
\end{align}
The finite cell--orbital parent is
\begin{equation}
 \widehat K_{\rho,N}^{(3)}
 =I_{N^3}\otimes T_0
 +\frac\rho3\sum_{\alpha=x,y,z}
 \left(\mathsf S_\alpha\otimes T_+
 +\mathsf S_\alpha^\dagger\otimes T_-\right).
 \label{eq:kronecker-parent}
\end{equation}
The tensor products act on lattice coordinates; the model retains four orbitals per
cell rather than introducing $4^3$ composite orbitals. Exact unitary conversion gives
\begin{equation}
 (\mathcal F_{3,N}\otimes I_4)\widehat K_{\rho,N}^{(3)}
 (\mathcal F_{3,N}^\dagger\otimes I_4)
 =\bigoplus_{i,j,k}K_\rho^{(3)}(\mathbf q_{ijk}).
 \label{eq:bloch-direct-sum}
\end{equation}
Denote the right-hand side by $\widehat K_{B,\rho,N}^{(3)}$. For complete unitary
fiber eigensystems
\begin{equation}
 K_\rho^{(3)}(\mathbf q_{ijk})
 =Q_{ijk}E_{ijk}Q_{ijk}^\dagger,
 \qquad E_{ijk}=\operatorname{diag}(\varepsilon_1,\ldots,\varepsilon_4),
\end{equation}
the conventional spectral representation is
\begin{equation}
 \widehat E_{\rho,N}^{(3)}=\bigoplus_{i,j,k}E_{ijk},
 \qquad
 \widehat K_{B,\rho,N}^{(3)}
 =\mathcal Q_{\rho,N}^{(3)}\widehat E_{\rho,N}^{(3)}
  (\mathcal Q_{\rho,N}^{(3)})^\dagger,
 \qquad \mathcal Q_{\rho,N}^{(3)}=\bigoplus_{i,j,k}Q_{ijk}.
 \label{eq:spectral-representation}
\end{equation}
The diagonal array $\widehat E$ without $\mathcal Q$ does not identify an operator in
orbital coordinates and may not be subtracted from a tight-binding Hamiltonian. The
verifier must reconstruct Eqs.~\eqref{eq:bloch-direct-sum} and
\eqref{eq:spectral-representation}, their inverses, and complete finite spectra without
requiring a dense $4N^3\times4N^3$ allocation.

A qualified frame defines
\begin{equation}
 \mathcal V_{\rho,N}^{(3)}=\bigoplus_{i,j,k}F_\rho^{(3)}(\mathbf q_{ijk})
\end{equation}
and
\begin{align}
 \widehat H_{B,\rho,N}^{(3)}
 &=(\mathcal V_{\rho,N}^{(3)})^\dagger
 \left[\bigoplus_{i,j,k}K_\rho^{(3)}(\mathbf q_{ijk})\right]
 \mathcal V_{\rho,N}^{(3)},\label{eq:retained-direct-sum}\\
 \widehat H_{W,\rho,N}^{(3)}
 &=(\mathcal F_{3,N}^\dagger\otimes I_2)
 \widehat H_{B,\rho,N}^{(3)}
 (\mathcal F_{3,N}\otimes I_2).
 \label{eq:wannier-transform}
\end{align}
The $2\times2$ block-circulant entries of $\widehat H_W$ are the finite Fourier hopping
blocks. This is the synthetic subspace-selection, gauge, and Bloch-to-Wannier chain; it
carries no material or first-principles claim.
